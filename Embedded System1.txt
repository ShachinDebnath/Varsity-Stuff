Exprement No. : 02(a)
Problem Name: Configure timer control registers of 8051 and develop a
program to generate delay.

#include<reg51.h>
void Delay(unsigned char count);
//timer delay function
void main(void)
{
TMOD = 0x01; while(1)
{
//Timer 0, mode 1 (16-bit)
P3 = 0x0F; Delay(10); P3 = 0xF0; Delay(10);
//low nibble ON
//10 x 50 ms = 0.5 s
//high nibble ON
}
}
void Delay(unsigned char count)
{
unsigned char n;
for(n = 0; n < count; n++)
{
TH0 = 0x3C; //50 ms at 12 MHz
TL0 = 0xB0;
TR0 = 1; //start Timer 0
while(TF0 == 0); //wait overflow
TR0 = 0; //stop Timer 0
TF0 = 0; //clear flag
}
}

Exprement No.: 04(b)
Problem Name: To demonstrate seven segments LED display and
generate counting from 0 to 99.

#include<reg51.h>
//hex values of digits 0-9 for 7-seg LED
//(common cathode)
unsigned char code digit[10] =
{0x3F, 0x06, 0x5B, 0x4F, 0x66,
0x6D, 0x7D, 0x07, 0x7F, 0x6F};
void delay(unsigned int ms)
{
unsigned int i, j;
for(i = 0; i < ms; i++)
{
for(j = 0; j < 120; j++)
{
}
}
}
void main(void)
{
unsigned char count;
P1 = 0x00;
P2 = 0x00;
while(1)
{
for(count = 0; count <= 99; count++)
{
P2 = digit[count / 10]; //tens
P1 = digit[count % 10]; //ones
delay(500);
}
}
}

Exprement No. : 05
Problem Name: Develop a program to interface stepper motors with
8051.

#include<reg51.h>
//full step sequence of the motor
unsigned char code step[4] =
{0x03, 0x06, 0x0C, 0x09};
void delay(unsigned int ms)
{
unsigned int i, j;
for(i = 0; i < ms; i++)
{
for(j = 0; j < 120; j++)
{
}
}
}
void main(void)
{
unsigned int n;
while(1)
{
for(n = 0; n < 200; n++)
{
P2 = step[n % 4]; delay(20);
//CW
}
delay(1000);
for(n = 0; n < 200; n++)
{
P2 = step[3 - (n % 4)]; //ACW
delay(20);
}
delay(1000);
}
}
